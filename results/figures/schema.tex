% NOTE: sn-jnl.cls inverts the usual size ordering. Measured against the class:
% footnotesize is 7pt, scriptsize is 9pt, tiny is 10pt. Sub-labels therefore cannot
% use the smaller-sounding commands -- they render larger than the 7pt node font and
% overflow their own shapes. subtxt fixes an explicit size below the node font.
\newcommand{\subtxt}{\fontsize{7.2}{8.2}\selectfont}
% Architecture schema of the AI-ASSIST system (TikZ).
% Adapted from paper-ijepes/figures/schema.tex on explicit instruction (ground-up
% policy otherwise applies to this paper - see CLAUDE.md). Checked against the
% current codebase (src/domain/estimation/service.py, src/services/qdrant/) on
% 2026-08-07 and updated where it had drifted:
%   - the feature-mapping box previously listed "domain-informed scaling" and
%     "zero-variance removal" as pipeline steps. Both are stale: the scaler
%     originally divided each feature group by a hand-picked range constant,
%     found to badly unbalance retrieval (one feature group could dominate
%     total distance by scaling accident - see paper-sr/EXPERIMENTS.md #16),
%     and was replaced by per-column data-driven standardization with no
%     separate variance-thresholding step (StandardScaler handles a constant
%     column safely on its own).
% Three layers: offline pipeline (top), shared stores (middle), online
% retrieval (bottom). The feature mapping phi is constructed offline and the
% *same* fitted mapping is applied to queries online. Zero-variance/near-
% duplicate columns are handled implicitly by standardization rather than by a
% separate pipeline step (see note above) - exact topology filter before
% similarity search still holds, as does the no-prediction-intervals caveat.
\begin{tikzpicture}[
    font=\fontsize{8.5}{10}\selectfont,
    node distance=3mm and 3mm,
    box/.style={draw, rounded corners=1pt, align=center, fill=gray!10,
                inner sep=4pt, minimum height=10mm, text width=22mm},
    proc/.style={draw, rounded corners=1pt, align=center, fill=blue!8,
                 inner sep=4pt, minimum height=10mm, text width=22mm},
    store/.style={draw, cylinder, shape border rotate=90, aspect=0.12,
                  align=center, fill=orange!15, inner sep=2pt,
                  text width=20mm, minimum height=15mm},
    outbox/.style={draw, rounded corners=1pt, align=center, fill=green!10,
                   inner sep=4pt, minimum height=10mm, text width=22mm},
    lab/.style={font=\subtxt\itshape, gray, fill=white, inner sep=1.4pt},
    arr/.style={-{Latex[length=2mm]}, semithick},
    qarr/.style={arr, dashed, blue!50!black},
]

% ---------------- Online stage (bottom row) ----------------
\node[box] (query) {Query pre-fault state\\(operational snapshot)};
\node[proc, right=of query] (phi) {Same mapping\\$\phi(\cdot)$};
\node[proc, right=of phi] (topo) {Exact topology\\filter $\psi(\tau)$};
\node[proc, right=of topo] (knn) {$K$-nearest-neighbor\\similarity search\\(Euclidean)};
\node[proc, right=of knn] (agg) {Record retrieval $+$\\distance-weighted\\aggregation};
\node[outbox, right=of agg] (dash) {CCT estimates,\\outcome distributions,\\neighborhood-support\\diagnostics};

\draw[qarr] (query) -- (phi);
\draw[qarr] (phi) -- (topo);
\draw[qarr] (topo) -- (knn);
\draw[qarr] (knn) -- (agg);
\draw[qarr] (agg) -- (dash);

% ---------------- Shared stores (middle layer) ----------------
\node[store, above=10mm of knn] (index) {Feature-vector\\index};
\node[store, text width=22mm, above=10mm of agg] (records)
    {Contingency-record\\archive\\{\subtxt states and outcomes}};

\draw[arr] (index.south) -- (knn.north);
\draw[arr] (records.south) -- (agg.north)
    node[lab, midway, right=1mm] {associated contingency records};

% ---------------- Offline stage (top row) ----------------
\node[proc, above=10mm of index] (prep) {Feature mapping $\phi$\\[1pt]
    {\subtxt sine--cosine angle encoding,}\\
    {\subtxt per-column standardization}};
\node[proc, left=of prep] (sim) {Offline time-domain\\simulation};
\node[box, left=of sim] (inputs) {Grid model and\\sampled pre-fault states};

\draw[arr] (inputs) -- (sim);
\draw[arr] (sim) -- (prep);
\draw[arr] (prep.south) -- (index.north)
    node[lab, pos=0.18, left=1mm] {feature vectors $z_i$};
\draw[arr] (prep.east) -| (records.north)
    node[lab, pos=0.25, above] {records};

% Same mapping reused online (constructed offline, applied to queries)
\draw[arr, dashed, gray] ([xshift=-10mm]prep.south) -- ++(0,-5mm)
    -| (phi.north);

% Stage labels
\node[lab, anchor=south west] at ([yshift=1mm]inputs.north west)
    {Offline: database construction};
\node[lab, anchor=north west] at ([yshift=-1mm]query.south west)
    {Online: retrieval-based inference (no new simulation, no calibrated intervals)};

\end{tikzpicture}
